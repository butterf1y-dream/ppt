3.1 소스 프로그램

#include <stdio.h>
#include <string.h>

int main() {
    char input[500];
    const char *keywords[] = {
        "ignore previous instructions", "system prompt",
        "developer mode", "jailbreak", "bypass",
        "ignore all rules", "reveal hidden instructions",
        "prompt injection"
    };
    int count = sizeof(keywords) / sizeof(keywords[0]);
    int detected = 0;
    printf("문장을 입력하세요: ");
    fgets(input, sizeof(input), stdin);
    for (int i = 0; i < count; i++) {
        if (strstr(input, keywords[i]) != NULL) { detected = 1; break; }
    }
    if (detected) printf("경고: 프롬프트 인젝션 관련 표현이 감지되어 입력이 거부되었습니다.\n");
    else printf("안전한 입력으로 판단되었습니다.\n");
    return 0;
}

3.2 동작 원리 설명

헤더파일

stdio.h — 입출력 / string.h — 문자열 검색(strstr)

input 배열

사용자 입력 문장을 500자까지 저장

keywords 배열

위험 문구 8개를 상수 문자열 포인터 배열로 저장

탐지 로직

for 반복문 + strstr()으로 키워드 존재 여부 순차 확인

detected 변수

위험 키워드 발견 시 1로 변경 → 경고 메시지 출력

안전 판정

키워드 미발견 시 안전 메시지 출력 후 종료
